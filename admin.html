async function createRetailer() { const message = document.getElementById("cm"); const button = document.getElementById("createBtn"); const full_name = document.getElementById("rn").value.trim(); const mobile = document.getElementById("rm").value.trim(); const password = document.getElementById("rp").value; let username = document.getElementById("ru").value.trim(); let retailer_number = document.getElementById("rr").value.trim(); if (!full_name) { msg(message, "Retailer full name is required.", true); return; } if (!/^\d{10}$/.test(mobile)) { msg( message, "Mobile number must be exactly 10 digits.", true ); return; } if (!username) { generateUsername(); username = document.getElementById("ru").value.trim(); } if (!retailer_number) { retailer_number = username; } if (!password) { msg( message, "Please enter retailer password.", true ); return; } if (password.length < 6) { msg( message, "Password must be at least 6 characters.", true ); return; } button.disabled = true; button.textContent = "⏳ Creating Retailer..."; try { const { data, error } = await supabaseClient.functions.invoke( "admin-retailer-action", { body: { action: "create", full_name: full_name, mobile: mobile, username: username, retailer_number: retailer_number, password: password } } ); console.log("CREATE RESULT:", data, error); if (error) { let errorText = error.message || "Failed to send request to Edge Function."; if ( error.context && typeof error.context.json === "function" ) { try { const server = await error.context.json(); if (server?.error) { errorText = server.error; if (server.details) { errorText += " - " + server.details; } } } catch (e) {} } msg(message, errorText, true); return; } if (!data || data.error) { msg( message, data?.error || "Retailer creation failed.", true ); return; } document.getElementById("showUser").value = data.username || username; document.getElementById("showPass").value = password; document .getElementById("credentialBox") .classList.remove("hidden"); msg( message, "Retailer created successfully." ); document.getElementById("rn").value = ""; document.getElementById("rm").value = ""; document.getElementById("rr").value = ""; document.getElementById("rp").value = ""; generateUsername(); await refreshDashboard(); } catch (error) { console.error("CREATE ERROR:", error); msg( message, error.message || "Retailer creation failed.", true ); } finally { button.disabled = false; button.textContent = "👤 Create Retailer"; } }
